cmake_minimum_required(VERSION 3.24)
project(perf_trace_tests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
find_package(Threads REQUIRED)
add_executable(ram-layout-test ../test_ram_layout.cpp)
target_include_directories(ram-layout-test PRIVATE ../../gpu/shadps4)
target_link_libraries(ram-layout-test PRIVATE Threads::Threads)
add_test(NAME ram-layout-test COMMAND ram-layout-test)
add_executable(staging-retention-test ../test_staging_retention.cpp)
target_include_directories(staging-retention-test PRIVATE ../../gpu/shadps4)
add_test(NAME staging-retention-test COMMAND staging-retention-test)
add_executable(perf-trace-test ../test_perf_trace.cpp ../../gpu/shim/tlg_perf_trace.cpp)
target_include_directories(perf-trace-test PRIVATE ../../gpu/shim ../../gpu/shadps4)
target_link_libraries(perf-trace-test PRIVATE Threads::Threads)

add_test(NAME perf-trace-test COMMAND perf-trace-test)
add_test(NAME perf-trace-shutdown COMMAND perf-trace-test shutdown)
add_test(NAME perf-trace-registration-loss COMMAND perf-trace-test registration)

add_test(NAME perf-trace-overflow-reason COMMAND perf-trace-test overflow)
